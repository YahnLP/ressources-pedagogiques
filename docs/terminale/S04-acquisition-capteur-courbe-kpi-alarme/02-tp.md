> **AGROSMART** vous confie la supervision thermique d'une **serre de tomates**. La température doit rester entre 18°C et 32°C. Au-delà de 32°C : alarme WARNING (chauffage peut être réduit). ## 🧪 PARTIE 1 — KPI Embarqués (30 min)

### Code à compléter : `TP_KPI_Serre.ino`

```cpp
#include <DHT.h>

#define DHT_PIN   4
#define DHT_TYPE  DHT22
#define LED_GREEN 25
#define LED_ORANGE 26
#define LED_RED   27

DHT dht(DHT_PIN, DHT_TYPE);

// ═══ KPI : Moyenne Mobile ══════════════════════════════════════════
const int N_MOYENNE = 8;
float  buf_moy[N_MOYENNE] = {0};
int    idx_moy   = 0;
bool   buf_plein = false;

float moyenneMobile(float nouvelle) {
  buf_moy[idx_moy] = nouvelle;
  idx_moy = (idx_moy + 1) % N_MOYENNE;
  if (idx_moy == 0) buf_plein = true;
  if (!buf_plein) return nouvelle;
  float s = 0;
  for (int i = 0; i < N_MOYENNE; i++) s += buf_moy[i];
  return s / N_MOYENNE;
}

// ═══ KPI : Min / Max glissants (N=20) ════════════════════════════
const int N_MINMAX = 20;
float  buf_mm[N_MINMAX] = {0};
int    idx_mm = 0;

void majMinMax(float val) {
  buf_mm[idx_mm] = val;
  idx_mm = (idx_mm + 1) % N_MINMAX;
}

float calculerMin() {
  float m = buf_mm[0];
  for (int i=1; i<N_MINMAX; i++) if(buf_mm[i]<m) m=buf_mm[i];
  return m;
}

float calculerMax() {
  float m = buf_mm[0];
  for (int i=1; i<N_MINMAX; i++) if(buf_mm[i]>m) m=buf_mm[i];
  return m;
}

// ═══ KPI : Tendance ══════════════════════════════════════════════
const int N_TENDANCE = 12;           // 12 points = 1 min (avec mesure/5s)
float  buf_tend[N_TENDANCE] = {0};
int    idx_tend = 0;
bool   tend_plein = false;

void majTendance(float val) {
  buf_tend[idx_tend] = val;
  idx_tend = (idx_tend + 1) % N_TENDANCE;
  if (idx_tend == 0) tend_plein = true;
}

float calculerTendance() {
  if (!tend_plein) return 0;
  int i_recent  = (idx_tend - 1 + N_TENDANCE) % N_TENDANCE;
  int i_ancien  = idx_tend;
  float delta   = buf_tend[i_recent] - buf_tend[i_ancien];
  float dt_h    = (N_TENDANCE - 1) * 5.0 / 3600.0;  // N points × 5s → heures
  return delta / dt_h;   // °C/h
}

// ═══ ALARMES ═════════════════════════════════════════════════════
// TODO PARTIE 2 : à compléter

// ═══ SETUP ═══════════════════════════════════════════════════════
void setup() {
  Serial.begin(115200);
  dht.begin();
  pinMode(LED_GREEN,  OUTPUT);
  pinMode(LED_ORANGE, OUTPUT);
  pinMode(LED_RED,    OUTPUT);
  digitalWrite(LED_GREEN, HIGH);   // Démarrage : tout vert
  Serial.println("ts_s,temp_raw,temp_moy,temp_min,temp_max,tendance_ch,alarme");
}

// ═══ LOOP ════════════════════════════════════════════════════════
void loop() {
  static unsigned long derniere_mesure = 0;
  if (millis() - derniere_mesure < 5000) return;
  derniere_mesure = millis();

  // Lecture capteur (ou simulation)
  float temp = dht.readTemperature();
  if (isnan(temp)) {
    // Simulation si pas de capteur :
    temp = 22.0 + sin(millis() / 30000.0) * 10.0 + random(-5, 5) / 10.0;
  }

  // Calcul KPI
  float moy   = moyenneMobile(temp);
  majMinMax(temp);
  float tmin  = calculerMin();
  float tmax  = calculerMax();
  majTendance(moy);
  float tend  = calculerTendance();

  // TODO PARTIE 2 : appliquer la logique d'alarme
  String alarme_str = "NORMAL";

  // Affichage CSV pour tableur
  Serial.printf("%.0f,%.1f,%.1f,%.1f,%.1f,%.2f,%s\n",
    millis()/1000.0, temp, moy, tmin, tmax, tend, alarme_str.c_str());
}
```

// ═══ ALARMES ═════════════════════════════════════════════════════
const float T_WARNING_H  = 32.0;    // Activation WARNING
const float T_WARNING_B  = 30.5;    // Désactivation WARNING (hystérésis 1,5°C)
const float T_CRITICAL_H = 38.0;    // Activation CRITICAL
const float T_CRITICAL_B = 36.5;    // Désactivation CRITICAL
const unsigned long ANTIREBOND_MS = ____;  // Q2.1 : à compléter

typedef enum { NORMAL, WARNING, CRITICAL } Niveau;
static Niveau niveau = NORMAL;

// Anti-rebond
static unsigned long debut_depassement = 0;
static bool en_depassement = false;
static Niveau niveau_en_attente = NORMAL;

Niveau evaluerNiveauHysteresis(float temp) {
  switch (niveau) {
    case NORMAL:
      if (temp >= T_CRITICAL_H) return CRITICAL;
      if (temp >= T_WARNING_H)  return WARNING;
      return NORMAL;
    case WARNING:
      if (temp >= T_CRITICAL_H)      return CRITICAL;
      if (temp < T_WARNING_B)        return NORMAL;
      return WARNING;
    case CRITICAL:
      if (temp < T_CRITICAL_B)       return WARNING;
      return CRITICAL;
  }
  return NORMAL;
}

void gererAlarme(float temp) {
  Niveau nouveau = evaluerNiveauHysteresis(temp);

  // Anti-rebond : confirmer seulement après ANTIREBOND_MS ms
  if (nouveau != niveau) {
    if (!en_depassement) {
      debut_depassement = millis();
      en_depassement = true;
      niveau_en_attente = nouveau;
    } else if (millis() - debut_depassement >= ANTIREBOND_MS
               && niveau_en_attente == nouveau) {
      niveau = nouveau;
      en_depassement = false;
    }
  } else {
    en_depassement = false;
  }
}

void majLEDs() {
  digitalWrite(LED_GREEN,  niveau == NORMAL   ? HIGH : LOW);
  digitalWrite(LED_ORANGE, niveau == WARNING  ? HIGH : LOW);
  digitalWrite(LED_RED,    niveau == CRITICAL ? HIGH : LOW);
}

const char* niveauStr(Niveau n) {
  if (n == WARNING)  return "WARNING";
  if (n == CRITICAL) return "CRITICAL";
  return "NORMAL";
}
