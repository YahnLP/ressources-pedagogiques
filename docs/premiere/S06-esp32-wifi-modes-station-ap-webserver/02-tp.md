## 🟢 NIVEAU 1 — Connexion WiFi en mode Station (15 min)

> Crée un nouveau sketch Arduino. **Projet : `S6_E2_WiFi_STA`**

```cpp
#include <WiFi.h>

const char* ssid     = "TON_SSID";      // ← À modifier
const char* password = "TON_PASSWORD";  // ← À modifier

void setup() {
  Serial.begin(115200);
  delay(1000);
  
  Serial.println("Démarrage...");
  WiFi.begin(ssid, password);
  
  // Attendre la connexion (max 20 secondes)
  int attempts = 0;
  while (WiFi.status() != WL_CONNECTED && attempts < 40) {
    delay(500);
    Serial.print(".");
    attempts++;
  }
  
  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\n✓ WiFi connecté !");
    Serial.print("Adresse IP : ");
    Serial.println(WiFi.localIP());
    Serial.print("Force signal (RSSI) : ");
    Serial.print(WiFi.RSSI());
    Serial.println(" dBm");
  } else {
    Serial.println("\n✗ Échec connexion WiFi");
    Serial.println("Vérifier SSID et mot de passe");
  }
}

void loop() {
  // Vérifier périodiquement l'état
  static unsigned long lastCheck = 0;
  if (millis() - lastCheck > 5000) {
    lastCheck = millis();
    Serial.print("Status WiFi : ");
    Serial.println(WiFi.status() == WL_CONNECTED ? "Connecté" : "Déconnecté");
  }
}
```

## 🟡 NIVEAU 2 — WebServer embarqué avec contrôle LED (25 min)

> Crée un nouveau sketch : **`S6_E2_WebServer`**

```cpp
#include <WiFi.h>
#include <WebServer.h>

const char* ssid     = "TON_SSID";
const char* password = "TON_PASSWORD";

const int LED_PIN = 2;          // LED intégrée sur GPIO 2
WebServer server(80);

// ── Page d'accueil ────────────────────────────────────────────────────────
void handleRoot() {
  String etat = digitalRead(LED_PIN) ? "ALLUMÉE 🟡" : "ÉTEINTE ⚫";
  
  String html = "<!DOCTYPE html><html><head>";
  html += "<meta charset='UTF-8'>";
  html += "<meta name='viewport' content='width=device-width, initial-scale=1'>";
  html += "<title>ESP32 CIEL</title>";
  html += "<style>body{font-family:Arial;text-align:center;margin:50px;}";
  html += "button{padding:15px 30px;font-size:18px;margin:10px;border:none;";
  html += "border-radius:8px;cursor:pointer;}";
  html += ".on{background:#4CAF50;color:white;}";
  html += ".off{background:#f44336;color:white;}</style></head><body>";
  html += "<h1>🔌 Contrôle ESP32</h1>";
  html += "<p>État LED : <strong>" + etat + "</strong></p>";
  html += "<a href='/led/on'><button class='on'>ALLUMER</button></a>";
  html += "<a href='/led/off'><button class='off'>ÉTEINDRE</button></a>";
  html += "<hr><p><small>IP : " + WiFi.localIP().toString() + "</small></p>";
  html += "</body></html>";
  
  server.send(200, "text/html", html);
}

// ── Commandes LED ─────────────────────────────────────────────────────────
void handleLedOn() {
  digitalWrite(LED_PIN, HIGH);
  server.sendHeader("Location", "/");       // Rediriger vers la page principale
  server.send(302, "text/plain", "");       // 302 = redirect
}

void handleLedOff() {
  digitalWrite(LED_PIN, LOW);
  server.sendHeader("Location", "/");
  server.send(302, "text/plain", "");
}

// ── API JSON simple ───────────────────────────────────────────────────────
void handleStatus() {
  String json = "{\"led\":" + String(digitalRead(LED_PIN) ? "true" : "false");
  json += ",\"ip\":\"" + WiFi.localIP().toString() + "\"";
  json += ",\"rssi\":" + String(WiFi.RSSI()) + "}";
  server.send(200, "application/json", json);
}

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);
  
  // Connexion WiFi
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.println("\nIP : " + WiFi.localIP().toString());
  
  // Enregistrer les routes
  server.on("/",          handleRoot);
  server.on("/led/on",    handleLedOn);
  server.on("/led/off",   handleLedOff);
  server.on("/status",    handleStatus);   // API JSON
  server.onNotFound([]() {
    server.send(404, "text/plain", "404 Not Found");
  });
  
  server.begin();
  Serial.println("Serveur HTTP démarré sur le port 80");
}

void loop() {
  server.handleClient();
}
```

## 🟠 NIVEAU 3 — Mode Access Point (15 min)

> Crée un nouveau sketch : **`S6_E2_ModeAP`**

```cpp
#include <WiFi.h>
#include <WebServer.h>

const char* ap_ssid = "ESP32_CIEL";     // Nom du réseau créé
const char* ap_pass = "ciel2024!";      // Min 8 caractères

WebServer server(80);

void setup() {
  Serial.begin(115200);
  
  // ── Mode Access Point ──────────────────────────────────────────────────
  WiFi.mode(WIFI_AP);
  bool ok = WiFi.softAP(ap_ssid, ap_pass);
  
  Serial.println(ok ? "AP créé !" : "Échec AP");
  Serial.println("SSID     : " + String(ap_ssid));
  Serial.println("IP portail: " + WiFi.softAPIP().toString());
  
  server.on("/", []() {
    int clients = WiFi.softAPgetStationNum();
    server.send(200, "text/html",
      "<html><head><meta charset='UTF-8'></head><body>"
      "<h1>🔧 Portail de configuration</h1>"
      "<p>Vous êtes connecté au réseau ESP32_CIEL</p>"
      "<p>Clients connectés : " + String(clients) + "</p>"
      "<form action='/save'>"
      "SSID réseau : <input name='ssid' placeholder='NomDuReseau'><br><br>"
      "Password    : <input name='pass' type='password'><br><br>"
      "<input type='submit' value='Enregistrer'>"
      "</form></body></html>");
  });
  
  server.on("/save", []() {
    String ssid = server.arg("ssid");
    String pass = server.arg("pass");
    Serial.println("SSID reçu : " + ssid);
    Serial.println("Pass reçu : " + pass);
    server.send(200, "text/html",
      "<h1>✓ Configuration reçue !</h1>"
      "<p>SSID : " + ssid + "</p>"
      "<p>(En production : sauvegarder en EEPROM et redémarrer)</p>");
  });
  
  server.begin();
}

void loop() {
  server.handleClient();
}
```

## 🔵 NIVEAU 4 — Client MQTT : Publisher et Subscriber (30 min)

> Crée un nouveau sketch : **`S6_E2_MQTT`**
> ⚠️ Bibliothèque requise : **PubSubClient** (Tools → Manage Libraries)

```cpp
#include <WiFi.h>
#include <PubSubClient.h>

// ── Configuration ─────────────────────────────────────────────────────────
const char* wifi_ssid   = "TON_SSID";
const char* wifi_pass   = "TON_PASSWORD";
const char* mqtt_broker = "broker.hivemq.com";  // ou IP du broker local
const int   mqtt_port   = 1883;

// !! Rendre le client_id unique : ajouter votre numéro de poste
const char* client_id   = "ESP32_CIEL_POSTE_XX"; // ← Remplacer XX

// ── Topics ────────────────────────────────────────────────────────────────
// Remplacer "XX" par votre numéro de poste
const char* TOPIC_TEMP = "ciel/labo/posteXX/temperature";
const char* TOPIC_LED  = "ciel/labo/posteXX/led";

// ── Objets ────────────────────────────────────────────────────────────────
WiFiClient   wifiClient;
PubSubClient mqtt(wifiClient);

// ── Callback MQTT : message reçu ──────────────────────────────────────────
void onMessage(char* topic, byte* payload, unsigned int len) {
  String msg = "";
  for (int i = 0; i < len; i++) msg += (char)payload[i];
  
  Serial.println("📩 [" + String(topic) + "] → " + msg);
  
  // Commande LED
  if (String(topic) == TOPIC_LED) {
    if (msg == "ON" || msg == "1") {
      digitalWrite(2, HIGH);
      Serial.println("💡 LED allumée");
    } else if (msg == "OFF" || msg == "0") {
      digitalWrite(2, LOW);
      Serial.println("💡 LED éteinte");
    }
  }
}

// ── Connexion MQTT (avec reconnexion) ──────────────────────────────────────
void reconnectMQTT() {
  while (!mqtt.connected()) {
    Serial.print("Connexion MQTT → ");
    if (mqtt.connect(client_id)) {
      Serial.println("✓ Connecté !");
      mqtt.subscribe(TOPIC_LED);   // S'abonner aux commandes LED
      Serial.println("Abonné : " + String(TOPIC_LED));
    } else {
      Serial.print("Échec (rc=");
      Serial.print(mqtt.state());
      Serial.println(") 5s...");
      delay(5000);
    }
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(2, OUTPUT);
  
  // WiFi
  WiFi.begin(wifi_ssid, wifi_pass);
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.println("\n✓ WiFi : " + WiFi.localIP().toString());
  
  // MQTT
  mqtt.setServer(mqtt_broker, mqtt_port);
  mqtt.setCallback(onMessage);
  reconnectMQTT();
}

void loop() {
  if (!mqtt.connected()) reconnectMQTT();
  mqtt.loop();
  
  // Publier la température toutes les 5 secondes
  static unsigned long lastPub = 0;
  if (millis() - lastPub > 5000) {
    lastPub = millis();
    // Température simulée entre 20 et 30°C
    float temp = 20.0 + (random(0, 100)) / 10.0;
    String payload = String(temp, 1);
    
    if (mqtt.publish(TOPIC_TEMP, payload.c_str())) {
      Serial.println("📤 Publié [" + String(TOPIC_TEMP) + "] : " + payload + "°C");
    }
  }
}
```
